\begin{column}{0.39\textwidth}
  {\Large\bfseries\color{byuOrange}Simpson's Paradox}\\[0.5em]
  \begin{itemize}
    \setlength\itemsep{0.55em}
    \item Each year has a clear positive relationship.
    \item The years contain different system generations and average scores.
    \item Combining them weakens the aggregate relationship.
  \end{itemize}
